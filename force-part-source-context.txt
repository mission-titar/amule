#include "ForcePartSelection.h"

namespace {
CMD4Hash s_fileHash;
uint32 s_part = 0;
bool s_active = false;
} // namespace

namespace ForcePartSelection {

void Set(const CMD4Hash &fileHash, uint32 part)
{
	s_fileHash = fileHash;
	s_part = part;
	s_active = true;
}

void Clear()
{
	s_fileHash.Clear();
	s_part = 0;
	s_active = false;
}

bool IsAllowed(const CMD4Hash &fileHash, uint32 part)
{
	return !s_active || (fileHash == s_fileHash && part == s_part);
}

} // namespace ForcePartSelection
